% ── En-tête / pied de page ───────────────────────────────────────────

\newcommand{\ficheHeaders}[3]{%
  \fancyhead[L]{\small\sffamily{\bfseries\textcolor{#3}{#1}}\hspace{0.7em}\textcolor{colSlate}{#2}}%

  \fancyfoot[C]{\small\sffamily\textcolor{colSlate}{\thepage\,/\,\pageref{LastPage}}}%

}

  \renewcommand{\headrule}{{\color{colRule}\hrule width\headwidth height 0.5pt}}%
}

% ── Bloc en-tête de fiche ────────────────────────────────────────────
% #1 code  #2 titre  #3 couleur domaine  #4 domaine  #5 ligne présence  #6 sous-thèmes
\newcommand{\ficheEntete}[6]{%
  \noindent{\color{#3}\rule{\linewidth}{3pt}}\par\vspace{7pt}
  \noindent{\sffamily\bfseries\LARGE\textcolor{#3}{#1}\hspace{0.7em}\textcolor{colInk}{#2}}\par\vspace{5pt}
  \noindent{\sffamily\small\textcolor{colSlate}{\MakeUppercase{#4}\quad\textbar\quad #5}}\par\vspace{2pt}
  \noindent{\sffamily\footnotesize\textcolor{colSlate}{Sous-thèmes : #6}}\par\vspace{5pt}
  \noindent{\color{colRule}\rule{\linewidth}{0.6pt}}\par\vspace{4pt}
}
