% ── Bloc en-tête de fiche : pavé à barre latérale épaisse ────────────
\newcommand{\ficheEntete}[6]{%
  \begin{tcolorbox}[enhanced, breakable,
    colback=#3!5!white,
    frame hidden, boxrule=0pt, arc=0mm,
    borderline west={4pt}{0pt}{#3},
    left=10pt, right=8pt, top=7pt, bottom=7pt,
    before skip=0pt, after skip=6pt]
    {\sffamily\bfseries\LARGE\textcolor{#3}{#1}\hspace{0.7em}\textcolor{colInk}{#2}}\\[4pt]
    {\sffamily\small\textcolor{colSlate}{\MakeUppercase{#4}\quad\textbar\quad #5}}\\[1pt]
    {\sffamily\footnotesize\textcolor{colSlate}{Sous-thèmes : #6}}
  \end{tcolorbox}%
}
